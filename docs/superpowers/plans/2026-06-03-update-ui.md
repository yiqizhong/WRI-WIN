# Update UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the hidden UpdateSection in Settings with a top-bar pill button + modal flow for detecting, downloading, and installing updates.

**Architecture:** `AppShell` subscribes to `window.writerUpdate.onState` and owns `updateState` + `modalOpen` booleans. It passes `updateState` + `onUpdateClick` to `WindowMasthead` (which renders the `UpdatePill`) and renders `<UpdateModal>` as a sibling. All update state comes from `AppUpdateState` in `@writer/contracts`.

**Tech Stack:** React, TypeScript, Tailwind CSS, Vitest, `WriterModal`, `useModalLatch`, `AppUpdateState`

```tsx
import { ArrowUp, Loader2 } from "lucide-react";
import type { AppUpdateState } from "@writer/contracts";
import { cn } from "@/lib/utils";

export function pillLabel(state: AppUpdateState): string | null {
  switch (state.status) {
    case "available":   return "Upgrade";
    case "downloading": return "Downloading…";
    case "downloaded":  return "Update ready";
    case "error":       return "Update failed";
    default:            return null;
  }
}

export type PillVariant = "blue" | "gray" | "green" | "warning";

export function pillVariant(state: AppUpdateState): PillVariant | null {
  switch (state.status) {
    case "available":   return "blue";
    case "downloading": return "gray";
    case "downloaded":  return "green";
    case "error":       return "warning";
    default:            return null;
  }
}

const VARIANT_CLASSES: Record<PillVariant, string> = {
  blue:    "bg-[var(--writer-accent-blue)] text-[#e5f1ff] hover:opacity-90",
  gray:    "bg-white/10 text-[#e5f1ff]/50 cursor-default",
  green:   "bg-[#1a7a4a] text-white hover:bg-[#166040]",
  warning: "bg-[#7a3010]/80 text-[#ffb89a] hover:bg-[#7a3010]",
};

type UpdatePillProps = {
  state: AppUpdateState;
  onClick: () => void;
};

export function UpdatePill({ state, onClick }: UpdatePillProps) {
  const label = pillLabel(state);
  const variant = pillVariant(state);
  if (!label || !variant) return null;

  const isDownloading = state.status === "downloading";

  return (
    <button
      type="button"
      onClick={isDownloading ? undefined : onClick}
      className={cn(
        "self-center mr-2 flex items-center gap-1.5 rounded-full px-3 py-1",
        "text-xs font-medium tracking-wide transition-opacity",
        VARIANT_CLASSES[variant]
      )}
      aria-label={label}
    >
      {state.status === "available"   && <ArrowUp   size={11} strokeWidth={2.5} aria-hidden />}
      {state.status === "downloading" && <Loader2   size={11} className="animate-spin"   aria-hidden />}
      {label}
    </button>
  );
}
```

```tsx
import type { ReactNode } from "react";
import type { AppUpdateState } from "@writer/contracts";
import { WriterModal } from "@/components/writer/writer-modal";
import { WriterPrimaryButton } from "@/components/writer/writer-primary-button";
import { useModalLatch } from "@/lib/use-modal-latch";

type UpdateModalProps = {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  state: AppUpdateState;
  onDownload: () => void;
  onRestart: () => void;
  onRetry: () => void;
};

export function UpdateModal({
  open,
  onOpenChange,
  state,
  onDownload,
  onRestart,
  onRetry,
}: UpdateModalProps) {
  const s = useModalLatch(open, state);

  return (
    <WriterModal
      open={open}
      onOpenChange={onOpenChange}
      title={resolveTitle(s)}
      footer={resolveFooter(s, { onOpenChange, onDownload, onRestart, onRetry })}
    >
      {resolveBody(s)}
    </WriterModal>
  );
}

function resolveTitle(s: AppUpdateState): string {
  switch (s.status) {
    case "available":   return `Writer ${s.version} is available`;
    case "downloading": return "Downloading update…";
    case "downloaded":  return "Update ready";
    case "error":       return "Update failed";
    default:            return "App updates";
  }
}

function resolveBody(s: AppUpdateState): ReactNode {
  const muted = "text-sm text-[#e5f1ff]/70";
  switch (s.status) {
    case "available":
      return (
        <p className={muted}>
          A new version of Writer is ready to download. This won't delete any of your data.
        </p>
      );
    case "downloading":
      return (
        <div className="flex flex-col gap-2">
          <div className="h-1.5 w-full overflow-hidden rounded-full bg-white/10">
            <div
              className="h-full rounded-full bg-[var(--writer-accent-blue)] transition-[width] duration-500 ease-linear"
              style={{ width: `${s.percent}%` }}
            />
          </div>
          <p className="text-xs text-[#e5f1ff]/50">{s.percent}%</p>
        </div>
      );
    case "downloaded":
      return <p className={muted}>Writer will restart and reopen automatically.</p>;
    case "error":
      return <p className={muted}>{s.message}</p>;
    default:
      return null;
  }
}

function SecondaryBtn({ onClick, children }: { onClick: () => void; children: ReactNode }) {
  return (
    <button
      type="button"
      onClick={onClick}
      className="px-4 py-2.5 text-sm font-medium text-[#e5f1ff]/60 transition-colors hover:text-[#e5f1ff]"
    >
      {children}
    </button>
  );
}

function resolveFooter(
  s: AppUpdateState,
  handlers: {
    onOpenChange: (open: boolean) => void;
    onDownload: () => void;
    onRestart: () => void;
    onRetry: () => void;
  }
): ReactNode {
  const { onOpenChange, onDownload, onRestart, onRetry } = handlers;
  const dismiss = () => onOpenChange(false);
  switch (s.status) {
    case "available":
      return (
        <>
          <SecondaryBtn onClick={dismiss}>Not now</SecondaryBtn>
          <WriterPrimaryButton onClick={onDownload}>Upgrade</WriterPrimaryButton>
        </>
      );
    case "downloading":
      return <SecondaryBtn onClick={dismiss}>Later</SecondaryBtn>;
    case "downloaded":
      return (
        <>
          <SecondaryBtn onClick={dismiss}>Later</SecondaryBtn>
          <WriterPrimaryButton onClick={onRestart}>Restart to install</WriterPrimaryButton>
        </>
      );
    case "error":
      return (
        <>
          <SecondaryBtn onClick={dismiss}>Dismiss</SecondaryBtn>
          <WriterPrimaryButton onClick={onRetry}>Try again</WriterPrimaryButton>
        </>
      );
    default:
      return null;
  }
}
```

- [ ] **Update `AppShell.tsx`** — subscribe to update state, own modal open/closed, render `UpdateModal`:

```tsx
import { type ReactNode, useEffect, useRef, useState } from "react";
import { useLocation, useNavigate } from "react-router-dom";
import type { AppUpdateState } from "@writer/contracts";

import "../styles/writer-app-frame.css";

import { WRITER_MODAL_HOST_ID } from "@/lib/writer-modal-host";
import { UpdateModal } from "@/components/writer/update-modal";

import { mastheadConfigForPath } from "./masthead-route";
import { WriterWindowBackground } from "./WriterWindowBackground";
import { WindowMasthead } from "./WindowMasthead";
import { BackdropTuner } from "./BackdropTuner";

type AppShellProps = {
  children: ReactNode;
  hideSettings?: boolean;
};

export function AppShell({ children, hideSettings }: AppShellProps) {
  const navigate = useNavigate();
  const { pathname } = useLocation();
  const { backTo } = mastheadConfigForPath(pathname);
  const onSettings = hideSettings || pathname === "/settings" ? undefined : () => navigate("/settings");

  const api = window.writerUpdate;
  const [updateState, setUpdateState] = useState<AppUpdateState>({ status: "idle" });
  const [modalOpen, setModalOpen] = useState(false);
  // Track whether we've auto-opened for this launch to avoid re-opening every re-render.
  const hasAutoOpened = useRef(false);

  useEffect(() => {
    if (!api) return;
    return api.onState((state) => {
      setUpdateState(state);
      // Auto-open modal on first available or error detection this session.
      if (!hasAutoOpened.current && (state.status === "available" || state.status === "error")) {
        hasAutoOpened.current = true;
        setModalOpen(true);
      }
    });
  }, [api]);

  return (
    <WriterWindowBackground>
      <div className="writer-app-viewport">
        <WindowMasthead
          {...(backTo ? { onBack: () => navigate(backTo) } : {})}
          {...(onSettings ? { onSettings } : {})}
          {...(api ? { updateState, onUpdateClick: () => setModalOpen(true) } : {})}
        />
        <div className="writer-app-viewport__content">{children}</div>
        <div id={WRITER_MODAL_HOST_ID} className="writer-modal-host" />
      </div>

      {api && (
        <UpdateModal
          open={modalOpen}
          onOpenChange={setModalOpen}
          state={updateState}
          onDownload={() => { void api.download().catch(() => undefined); }}
          onRestart={() => { void api.restartToInstall().catch(() => undefined); }}
          onRetry={() => { void api.check().catch(() => undefined); }}
        />
      )}

      <BackdropTuner />
    </WriterWindowBackground>
  );
}
```
